 // Tarjeta visual interactiva para seleccionar una asignatura
 @Composable
 fun SubjectCard(
     subject: Subject,
     onClick: () -> Unit
 ) {
     Card(
         modifier = Modifier
             .fillMaxWidth()
             .padding(8.dp)

             Text(

                 style = MaterialTheme.typography.bodyMedium
             )
         }
     }
 }
