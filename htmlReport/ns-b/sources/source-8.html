 @Composable
 fun SubjectSelectionScreen(
     subjects: List<Subject>,
     attemptsPerSubject: Map<String, Int>,
     availableSubjectIds: Set<String>,
     onSubjectSelect: (Subject) -> Unit,
     onLogout: () -> Unit
 ) {
     Surface(
         modifier = Modifier.fillMaxSize(),
         color = Color(0xFFF4F9FC)
     ) {
         Column(
             modifier = Modifier
                 .fillMaxSize()
                 .padding(16.dp)
         ) {
             // Cabecera Infantil
             Row(
                 modifier = Modifier
                     .fillMaxWidth()
                     .padding(vertical = 8.dp),
                 horizontalArrangement = Arrangement.SpaceBetween,
                 verticalAlignment = Alignment.CenterVertically
             ) {
                 Column {
                     Text(
                         text = "¡Hola, Estudiante! 👋",
                         fontSize = 22.sp,
                         fontWeight = FontWeight.Black,
                         color = Color(0xFF1E293B)
                     )
                     Text(
                         text = "¿Qué desafío resolveremos hoy?",
                         fontSize = 14.sp,
                         color = Color.Gray
                     )
                 }
 
                 Button(
                     onClick = onLogout,
                     colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFF6B6B)),
                     shape = RoundedCornerShape(14.dp)
                 ) {
                     Text("Salir", fontWeight = FontWeight.Bold)
                 }
             }
 
             Spacer(modifier = Modifier.height(12.dp))
 
             // Lista de Asignaturas
             LazyColumn {
                 items(subjects) { subject ->
                     val attemptsDone = attemptsPerSubject[subject.title] ?: 0
                     val isCompleted = attemptsDone >= 2
 
                     // Solo se puede entrar si UTP autorizó al menos un OA de la asignatura
                     val hasContent = subject.id in availableSubjectIds
                     val isLocked = isCompleted || !hasContent
 
                     // Icono y color temático por asignatura
                     val (emoji, cardAccent) = when (subject.title) {
                         "Matemática" -> Pair("🧮", Color(0xFF00A8E8))
                         "Lenguaje y Comunicación" -> Pair("📚", Color(0xFFFF7F3F))
                         "Ciencias Naturales" -> Pair("🔬", Color(0xFF06D6A0))
                         else -> Pair("🗺️", Color(0xFFEF476F))
                     }
 
                     Card(
                         onClick = { if (!isLocked) onSubjectSelect(subject) },
                         enabled = !isLocked,
                         shape = RoundedCornerShape(22.dp),
                         colors = CardDefaults.cardColors(
                             containerColor = if (isLocked) Color(0xFFE2E8F0) else Color.White
                         ),
                         elevation = CardDefaults.cardElevation(defaultElevation = if (isLocked) 1.dp else 4.dp),
                         modifier = Modifier
                             .fillMaxWidth()
                             .padding(vertical = 8.dp)
                             .alpha(if (isLocked) 0.65f else 1.0f)
                     ) {
                         Row(
                             modifier = Modifier
                                 .fillMaxWidth()
                                 .padding(16.dp),
                             verticalAlignment = Alignment.CenterVertically
                         ) {
                             // Avatar con Emoji de la Asignatura
                             Box(
                                 modifier = Modifier
                                     .size(56.dp)
                                     .background(cardAccent.copy(alpha = 0.15f), CircleShape),
                                 contentAlignment = Alignment.Center
                             ) {
                                 Text(text = emoji, fontSize = 28.sp)
                             }
 
                             Spacer(modifier = Modifier.width(14.dp))
 
                             Column(modifier = Modifier.weight(1f)) {
                                 Text(
                                     text = subject.title,
                                     fontSize = 17.sp,
                                     fontWeight = FontWeight.Bold,
                                     color = Color(0xFF1E293B)
                                 )
                                 Spacer(modifier = Modifier.height(2.dp))
                                 Text(
                                     text = subject.description,
                                     fontSize = 12.sp,
                                     color = Color.Gray
                                 )
 
                                 Spacer(modifier = Modifier.height(8.dp))
 
                                 // Badge de Intentos (Chip)
                                 Surface(
                                     color = if (isCompleted) MaterialTheme.colorScheme.error.copy(alpha = 0.1f) else cardAccent.copy(alpha = 0.15f),
                                     shape = RoundedCornerShape(10.dp)
                                 ) {
                                     Text(
                                         text = if (!hasContent) {
                                             "⏳ Esperando autorización de UTP"
                                         } else if (isCompleted) {
                                             "🔒 Intentos agotados (2/2)"
                                         } else {
                                             "⭐ Intentos realizados: $attemptsDone/2"
                                         },
                                         fontSize = 11.sp,
                                         fontWeight = FontWeight.Bold,
                                         color = if (isCompleted) MaterialTheme.colorScheme.error else cardAccent,
                                         modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp)
                                     )
                                 }
                             }
 
                             if (isLocked) {
                                 Icon(
                                     imageVector = Icons.Filled.Lock,
                                     contentDescription = "Bloqueado",
                                     tint = Color.Gray,
                                     modifier = Modifier.size(24.dp)
                                 )
                             }
                         }
                     }
                 }
             }
         }
     }
 }
